do_all_renames = {  #自定义效果，进行所有命名
	every_state = {
		### Chinese ###
		if = {
			limit = { #判断省份控制国
				controller = { is_independent_china_or_warlord = yes }
			}
			do_chinese_renames_in_asia = yes #进行对应的重命名
		}
		else_if = {
			limit = { controller = { tag = RYU } }
			do_ryukyuan_renames_in_asia = yes
		}
		else_if = {
			limit = { controller = { tag = JAP } }
			do_japanese_renames = yes
		}
		### France ###
		else_if = {
			limit = {
				controller = { tag = FRA }
			}
			do_france_renames = yes
		}
		else_if = {
			limit = {
				controller = { tag = BRI }
			}
			do_breizh_renames_in_breizh = yes
		}
		### Germany ###
		else_if = {
			limit = {
				controller = {
					tag = GER
					has_government = fascism
				}
			}
			do_nazi_germany_renames = yes
		}
		### Greece ###
		else_if = {
			limit = {
				controller = { tag = GRE }
			}
			do_greek_renames = yes
		}

		### Poland ###
		else_if = {
			limit = {
				controller = { tag = POL }
			}
			do_polish_renames = yes
		}

		### Ottoman ###
		else_if = {
			limit = {
				controller = {
					tag = TUR
					has_completed_focus = TUR_return_of_the_sultan
				}				
			}
			do_ottoman_renames = yes
		}
	}
}

三，决议类别decisions
HJM_electrify_state = {
    icon = decision_generic_electricity  -- 决议图标  
	  target_trigger = {  -- 目标州需要满足的条件
        FROM = {
            has_state_flag = A_flag  -- 必须有“”标记
            NOT = { has_state_flag = A_flag }  -- 还没有
            is_fully_controlled_by = ROOT  -- 完全由本国控制
        }
    }
    state_target = any_controlled_state  -- 目标可以是任何被控制的州
    on_map_mode = map_only  -- 只能在地图模式下触发
    fire_only_once = yes  -- 每个州只能执行一次
    cost = 25  -- 花费25政治点数
    modifier = {  -- 执行期间的国家 modifier
        consumer_goods_factor = 0.10  -- 消费品需求增加10%
    }
    remove_effect = {  -- 决议完成后的效果
        maodiehaqi_effect = yes  -- 触发一个自定义效果（哈气）
    }
    ai_will_do = {  -- AI执行意愿
        base = 125  -- 基础权重
        modifier = {
            factor = 2  -- 和平时期权重翻倍
            has_war = no 
        }
        modifier = {
            factor = 0  -- 投降进度超过20%时不执行
            surrender_progress > 0.2
        }
    }
}

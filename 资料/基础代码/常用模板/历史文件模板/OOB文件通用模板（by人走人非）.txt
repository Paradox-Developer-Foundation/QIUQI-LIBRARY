# 1.3 开局生产线（instant_effect）
# ==============================
# instant_effect 块定义开局即刻生效的生产线设置
# 这些生产线会被提取到国家历史文件中（按 MOD 规范）
# 注意：在国家历史文件中，creator 和 version_name 应省略（避免 equipmentvariant.cpp:1103 错误）

instant_effect = {
	add_equipment_production = {					# 每条生产线定义一种装备的生产
		equipment = {
			type = infantry_equipment_1				# 装备类型（必须与 common/units/equipment/ 中的定义一致）
			creator = TAG							# 装备设计者（TAG 大写，国家历史文件中应省略）
		}
		requested_factories = 3						# 分配工厂数量（整数）
		progress = 0.1								# 初始生产进度（0.0~1.0，0.1 = 10%）
		efficiency = 50								# 初始生产效率（0~100，50 = 50%）
		industrial_manufacturer = mio:TAG_manufacturer_organization	# 工业制造商（可选，需要相应 DLC）
	}

	# 一条生产线可以同时生产多个国家标记的装备：
	add_equipment_production = {
		equipment = {
			type = infantry_equipment_1
			creator = TAG1
		}
		equipment = {
			type = infantry_equipment_1
			creator = TAG2
		}
		requested_factories = 2
		progress = 0.1
		efficiency = 50
	}
}

# 适用场景：拥有 TAOG（Arms Against Tyranny，以武抗暴）DLC 时的陆军编制
# TAOG 引入了"团级支援连"（Regimental Support Companies）系统，在师级支援连之外新增独立的支援连槽位
# 命名规则：与基础版/NSB版共用同一文件，通过 regimental_support 块区分

# ==============================
# 2B.1 团级支援连 vs 师级支援连
# ==============================
# ┌─────────────────┬──────────────────────┬──────────────────────┐
# │ 属性            │ 师级支援连 (support)  │ 团级支援连 (regimental_support) │
# ├─────────────────┼──────────────────────┼──────────────────────┤
# │ 槽位系统        │ 5个固定槽位          │ 独立槽位（与师级分开）│
# │ 战斗宽度        │ 0                     │ 0                     │
# │ 单位定义参数    │ group = support       │ group = support       │
# │                 │ divisional = yes      │ divisional = no       │
# │ 定义文件        │ common/units/ 各文件  │ common/units/fire_support.txt │
# │ 模板中位置      │ regiments 之后        │ regiments 之后、support 之前 │
# │ 坐标规则        │ x = 0, y 递增         │ x = 1, y 递增         │
# │ 解锁方式        │ 科技自动解锁          │ 部分需 unlock_subunit │
# └─────────────────┴──────────────────────┴──────────────────────┘

# ==============================
# 2B.2 团级支援连完整列表
# ==============================
# 默认可用（active = yes，开局即可使用）：
#   field_guns          = 团级野战炮        | 软攻+，需要火炮装备
#   anti_air_battery    = 团级防空排        | 对空攻击+，需要防空装备
#   anti_tank_battery   = 团级反坦克排      | 穿甲+，需要反坦克装备
#   rocket_battery      = 团级火箭炮排      | 软攻+，需要火箭炮装备

# 需解锁（active = no，需通过 unlock_subunit 解锁）：
#   fire_support        = 步兵火力支援连    | 通用火力支援，需要步兵装备+支援装备
#   mot_fire_support    = 摩托化火力支援连  | 摩托化版火力支援，需要步兵装备+支援装备+摩托化装备
#   mot_field_guns      = 摩托化团级野战炮  | 摩托化版野战炮
#   pack_artillery      = 驮载炮兵          | 山地/特种部队适用

# =================================================================================================
#                                      第九部分：常见错误与注意事项
# =================================================================================================
# 1. division_template 必须在 units 块之前定义，否则游戏无法找到模板
# 2. 国家历史文件中引用的 division_template 必须在此 OOB 文件中声明
# 3. 舰船名称必须唯一，同一国家内不能有重复的舰船名
# 4. 省份ID必须是 1-9399 范围内的陆地省份（海军基地使用州ID）
# 5. NSB 下 force_equipment_variants 必须使用 chassis 类型 + version_name
# 6. Legacy 海军装备类型（battleship_1 等）与 MTG 类型（ship_hull_*）不能混用
# 7. 空军 OOB 中的 equipment 类型必须与 BBA/Legacy 版本一致
# 8. 文件末尾必须保留至少一行空行，否则最后一行可能不被读取
# 9. 使用 UTF-8 无 BOM 编码
# 10. 营的坐标 (x, y) 不要重复，同一位置不能有多个营
# 11. 支援连的 x 坐标始终为 0，y 坐标连续递增
# 12. division_names_group 必须在 common/units/names/ 中有对应的名列表文件
# 13. force_equipment_variants 中的 version_name 必须对应 create_equipment_variant 中的 name
# 14. 在国家历史文件中的 add_equipment_production 不要使用 creator 和 version_name（避免 equipmentvariant.cpp:1103 错误）
# 15. regimental_support 块必须放在 regiments 和 support 之间，写错位置会导致模板解析失败
# 16. 团级支援连的坐标 (x, y) 不要与师级支援连的坐标重复（regimental_support 使用 x=1 列，support 使用 x=0 列）
# 17. 团级支援连中 active = no 的单位（如 fire_support、mot_fire_support 等）需要在国家历史文件中通过 unlock_subunit 解锁
# 18. AI 模板中的 regimental_support 使用"数量"语法（如 field_guns = 2），而非坐标语法（如 field_guns = { x = 1 y = 0 }）
# 19. 团级支援连（divisional = no）与 combat_support 组前线营（如 artillery_brigade，有战斗宽度）是不同的概念，不要混淆
# 20. TAOG DLC 的团级支援连系统与 NSB 坦克设计师、MTG 舰船设计师、BBA 飞机设计师都是独立系统，可任意组合使用
# 21. 无 TAOG DLC 时，regimental_support 块会导致解析错误——如需兼容无 TAOG 用户，应将 regimental_support 放在 has_dlc = "Arms Against Tyranny" 条件分支中
# 22. 同一 regimental_support 块内的多个团级支援连坐标 (x, y) 不能重复（与 regiments 坐标规则相同）

scripted_diplomatic_actions = { ＃来定制你的要求解散军队选项吧
	DIP_ACTION_DISARM = {
	allowed = { 
	always = yes
	}
	visible = { 
	always = yes
	}
	visible = { 
	always = yes
	}

	send_description = DIP_ACTION_DISARM_SEND_DESC
	receive_description = DIP_ACTION_DISARM_RECEIVE_DESC
	accept_title = DIP_ACTION_DISARM_ACCEPT_TITLE
	accept_description = DIP_ACTION_DISARM_ACCEPT_DESC
	reject_title = DIP_ACTION_DISARM_REJECT_TITLE
	reject_description = DIP_ACTION_DISARM_REJECT_DESC

	requires_acceptance = yes

	#行动发送时运行
	on_sent_effect = {
	}	
	# 行动成功发送并被接受时运行
	complete_effect = { ＃目标国删除所有军队和舰船
		THIS = {
			delete_unit = all
			destroy_ships = all
		}
	}	
	# 行动发送但被拒绝时运行
	reject_effect = {
		ROOT = {
			create_wargoal = {
    		type = anytype ＃填一个战争借口，只要在wargoal中定义了
    		target = PREV.THIS
		}
	}
	}
	}

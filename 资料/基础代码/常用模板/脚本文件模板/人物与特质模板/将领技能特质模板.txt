#国家领导人特质模板
#国家领导人特质保存在Hearts of Iron IV/common/country_leader/下的任意txt文件中，国家领导人特质可用于顾问、理念和领导人
leader_traits = { #特质文件的主结构，使游戏能够识别和应用特质
    my_trait = { #特质的id
        random = no #是否随机应用特质，如果是，则该特质可以出现在随机领导人身上
        sprite = 1 #决定在顾问左下角出现的该特质的贴图的帧。关于所有领导人特质的贴图存放在由Hearts of Iron IV/interface/ideas.gfx文件所注册的的帧图（GFX_idea_traits_strip）里。最多只能有一张帧图。
		#修正
        political_power_gain = 0.1

            }
        }
        ai_will_do = { #AI选择拥有该特质的顾问的概率

        }
        command_cap = 10 #具有该特质的顾问占用的指挥点数的数量。
    }
}
#单位领导人特质模板
#单位领导人特质保存在Hearts of Iron IV/common/unit_leader/下的任意txt文件中，单位领导人特质可用于单位领导人和特工。
leader_traits = {
 	my_trait = {

 		type = <type> # 适用身份类型，可选值：all（全部将领）、field_marshal（陆军元帅）、corps_commander（陆军将军）、navy（海军将领），operative（特工）
		trait_type = <trait_type> # 特质大类，可选参数：#basic_trait 基础特质，#status_trait 状态特质，#personality_trait 性格特质，#assignable_trait 可手动指派特质，#basic_terrain_trait 基础地形特质，#assignable_terrain_trait 可指派地形特质，#exile 流亡特质
		cost = <value> #升级所需的经验值

        #条件
		allowed = { 
			<trigger>
		} # 允许获取该特质的判定的条件，作用域：陆军将领 unit_leader
		prerequisites = { 
			<trigger>
		} # 解锁前置条件的条件，作用域：将领 unit leader
		gain_xp = { 
			<trigger>
		} # 战斗获取经验的条件，作用域：作战单位 combatant
		gain_xp_leader = { 
			<trigger>
		} # 将领自身获取经验的条件，作用域：将领；ROOT=所属国家，FROM=特工目标国籍

		override_effect_tooltip = <tooltip>_tt # 覆盖默认的提示，填写本地化Key。如，该特质为+1进攻时，一般来说会自动显示“+1进攻”，添加自定义提示后，会使用自定义的本地化值覆盖默认的自动提示。
		custom_effect_tooltip = <tooltip>_tt # 自定义效果提示本地化Key，附加在默认提示上
		custom_prerequisite_tooltip = <tooltip>_tt # 自定义解锁前置条件提示本地化Key，用于特质树
		custom_gain_xp_trigger_tooltip = <tooltip>_tt # 自定义获取经验触发提示本地化Key，用于特质树

		custom_prerequisite_tooltip = <tooltip>_tt

		genius_advisor_trait = <trait_id_3> #定义奇才的提示符

		unit_type = { #定义该特质适用于哪个units
			type = <unit_type>
			type = <unit_type>
		}

		unit_trigger = { #定义在何种条件下，特质的修正才会生效
			#对指挥的军队所满足的条件
			owner = {
				#对拥有这个特质的将领的国家所要满足的条件
			}
		}

        
		#特质树内容
		mutually_exclusive = other_trait # 互斥特质，填写冲突特质ID，二选一无法共存，用于特质树

		    traits = {XXX YYY ZZZ}  # 父特质ID列表

   		}

		

   		any_parent = { XXX YYY ZZZ } # 简写：任意满足列表中一个父特质（等价于num_parents_needed = 1），用于特质树
   		all_parents = { XXX YYY ZZZ } # 简写：必须拥有列表全部父特质才可解锁，用于特质树

		gui_row = 0 # 特质树UI行号（从0开始），留空或设为-1：不在可解锁特质树显示
		gui_column = 0 # 特质树UI列号（从0开始），留空或设为-1：自动排布列位置

		
		#修正与效果
		modifier = { ... } # 特质赋予指挥的师的修正
		non_shared_modifier = { ... } # 非共享修正,与modifier类似，只是在工具提示上有所不同。
		corps_commander_modifier = { ... } # 仅担任陆军将领时生效的修正

		new_commander_weight = { ... } # 游戏随机刷将领自带此特质的权重，如factor = 0为随机将领不会自动获得

scr_dec_get_btn_select_frame = {
	base = 0
	modifier = {
		set_temp_variable = {
			t_frame = 0
		}
		if = {
			limit = {
				var:scripted_decision_entry = {
					check_variable = {
						decision_status = global.scripted_decision_is_removing
					}
				}
			}
			set_temp_variable = {
				t_frame = 4
			}
		}
		else_if = {
			limit = {
				var:scripted_decision_entry = {
					OR = {
						check_variable = {
							decision_status = global.scripted_decision_has_removed
						}
						check_variable = {
							decision_status = global.scripted_mission_is_failed
						}
					}
				}
			}
			set_temp_variable = {
				t_frame = 3
			}
		}
		else_if = {
			limit = {
				scripted_decision_available_trigger = yes
			}
			set_temp_variable = {
				t_frame = 2
			}
		}
		else = {
			set_temp_variable = {
				t_frame = 1
			}
		}
		add = t_frame
	}
}

scr_dec_progressGetFrame = {
	base = 0
	modifier = {
		set_temp_variable = {
			t_percent = 0
		}
		var:scripted_decision_entry = {
			if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_is_removing
					}
				}
				set_temp_variable = {
					t_max = days_remove
				}
				clamp_temp_variable = {
					var = t_max
					min = 0
				}
				set_temp_variable = {
					t_current = days_remove
				}
				subtract_from_temp_variable = {
					t_current = days_remove_left
				}
				clamp_temp_variable = {
					var = t_current
					min = 0
					max = t_max
				}
				if = {
					limit = {
						check_variable = {
							t_max = 0
						}
					}
					if = {
						limit = {
							check_variable = {
								t_current > 0
							}
						}
						set_temp_variable = {
							t_percent = 0
						}
						else = {
							set_temp_variable = {
								t_percent = 100
							}
						}
					}
				}
				else = {
					set_temp_variable = {
						t_percent = t_current
					}
					multiply_temp_variable = {
						t_percent = 100
					}
					divide_temp_variable = {
						t_percent = t_max
					}
					round_temp_variable = t_percent
				}
				## 如果最初为0，则降低到0。-100
				##

			}
			else = {
				set_temp_variable = {
					t_max = days_mission_timeout
				}
				clamp_temp_variable = {
					var = t_max
					min = 0
				}
				set_temp_variable = {
					t_current = days_mission_timeout
				}
				subtract_from_temp_variable = {
					t_current = days_mission_timeout_left
				}
				clamp_temp_variable = {
					var = t_current
					min = 0
					max = t_max
				}
				if = {
					limit = {
						check_variable = {
							t_max = 0
						}
					}
					if = {
						limit = {
							check_variable = {
								t_current > 0
							}
						}
						set_temp_variable = {
							t_percent = 0
						}
						else = {
							set_temp_variable = {
								t_percent = 100
							}
						}
					}
				}
				else = {
					set_temp_variable = {
						t_percent = t_current
					}
					multiply_temp_variable = {
						t_percent = 100
					}
					divide_temp_variable = {
						t_percent = t_max
					}
					round_temp_variable = t_percent
				}

			}
		}
		add = t_percent
	}
}

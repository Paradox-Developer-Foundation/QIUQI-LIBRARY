scripted_decision_gui_default_slider_counter = {
	base = 0
	modifier = {
		check_variable = {
			scr_dec_decision_slider_height < 200
		}
		add_to_temp_array = {
			scr_dec_decision_slider_grid_array = scr_dec_decision_slider_height
		}
		add = scr_dec_decision_slider_height
	}
	modifier = {
		NOT = {
			check_variable = {
				scr_dec_decision_slider_height < 200
			}
		}
		subtract_from_temp_variable = {
			scr_dec_decision_slider_height = 100
		}
		add_to_temp_array = {
			scr_dec_decision_slider_grid_array = 100
		}
		add = mtth:scripted_decision_gui_default_slider_counter
	}
}

		set_temp_variable = {
			scr_dec_gui_y_base = 0
		}

		}

			array = scripted_decision_category_temp_list
			value = scripted_decision_category_entry
			index = i
			set_temp_variable = {
				scr_dec_decision_height = mtth:scripted_decision_category_default_height_counter
			}
		}
		set_temp_variable = {
			scr_dec_decision_gui_height = scr_dec_decision_category_height_sum
		}
		set_temp_variable = {
			scr_dec_decision_slider_height = scr_dec_decision_gui_height
		}
		clear_temp_array = scr_dec_decision_slider_grid_array
		set_temp_variable = {
			scr_dec_decision_slider_last_height = mtth:scripted_decision_gui_default_slider_counter
		}
		set_temp_variable = {
			scr_dec_decision_category_height_sum = 0 
		}
		log = "[?scr_dec_decision_slider_grid_array^num]"
		log = "[?scr_dec_decision_slider_last_height]"
		add = scr_dec_decision_gui_height
	}
}

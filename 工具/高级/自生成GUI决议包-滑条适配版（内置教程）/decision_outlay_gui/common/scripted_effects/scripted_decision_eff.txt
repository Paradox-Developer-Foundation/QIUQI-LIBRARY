scripted_decision_initiate = {
	if = {
		limit = {
			NOT = {
				has_global_flag = scripted_decision_all_initiated
			}
		}
		set_variable = {
			global.scripted_decision_last_id = -1
		}
		set_variable = {
			global.scripted_decision_icon_last_id = -1
		}
		set_variable = {
			global.scripted_decision_cost_last_id = -1
		}



		set_variable = {
			global.scripted_decision_default_status = 0
		}
		set_variable = {
			global.scripted_decision_is_removing = 1
		}
		set_variable = {
			global.scripted_decision_has_removed = 2
		}
		set_variable = {
			global.scripted_mission_is_active = 3
		}
		set_variable = {
			global.scripted_mission_is_failed = 4
		}

		clear_array = all_scripted_decision_lists
		
		set_global_flag = scripted_decision_all_initiated

		
		set_variable = {
			global.scripted_decision_category_last_id = -1
		}
		set_variable = {
			global.scripted_decision_category_icon_last_id = -1
		}
		
		set_variable = {
			global.scripted_decision_category_collapse = 1 
		}
		set_variable = {
			global.scripted_decision_category_show = 0
		}

		set_global_flag = scripted_decision_all_category_initiated
	}
}

		}
		set_variable = {
			days_remove = 0
		}
		set_variable = {
			decision_status = 0
		}
		set_variable = {
			days_mission_timeout = 0
		}
		set_variable = {
			days_re_enable = 0
		}
		set_variable = {
			priority = 0
		}
		var:icon = {
			set_character_name = "GFX_decision_unknown"
		}
		var:custom_cost_text = {
			set_character_name = " scripted_custom_cost_text_default_cost_text"
		}

scripted_decision_complete_scripted_decision_effect = {
	if = {
		limit = {
			scripted_decision_available_trigger = yes
			var:scripted_decision_entry = {
				NOT = {
					check_variable = {
						decision_status = global.scripted_decision_is_removing
					}
					check_variable = {
						decision_status = global.scripted_decision_has_removed
					}
				}
			}
		}
		set_temp_variable = {
			t_pp = -1
		}
		var:scripted_decision_entry = {
			multiply_temp_variable = {
				t_pp = cost
			}
		}
		add_political_power = t_pp
		var:scripted_decision_entry = {
			complete_scripted_decision = yes
		}
	}
}
scripted_decision_start_remove = {
	if = {
		limit = {
			scripted_decision_can_be_complete = yes
		}
		set_variable = {
			decision_status = global.scripted_decision_is_removing
		}
		set_variable = {
			days_remove_left = days_remove
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						days_remove > 0
					}
				}
			}
			remove_scripted_decision = yes
		}
	}
}
complete_scripted_decision = {
	meta_effect = {
		text = {
			if = {
				limit = {
					scripted_decision_can_be_complete = yes
				}
				owner = {
					[t_decision]_complete_effect = yes
				}
				scripted_decision_start_remove = yes
			}
		}
		t_decision = "[?THIS.GetName]"
	}
}
remove_scripted_decision = {
	meta_effect = {
		text = {
			if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_is_removing
					}
				}
				owner = {
					[t_decision]_remove_effect = yes
				}
				scripted_decision_end_remove = yes
			}
		}
		t_decision = "[?THIS.GetName]"
	}
}
scripted_decision_start_re_enable = {
	set_variable = {
		days_re_enable_left = days_re_enable
	}
	set_character_flag = scripted_decision_is_re_enabling
}
scripted_decision_end_remove = {
	if = {
		limit = {
			check_variable = {
				decision_status = global.scripted_decision_is_removing
			}
		}
		set_variable = {
			decision_status = global.scripted_decision_has_removed
		}
		scripted_decision_start_re_enable = yes
	}
}
activate_scripted_mission = {
	if = {
		limit = {
			scripted_decision_is_mission = yes
			OR = {
				check_variable = {
					decision_status = global.scripted_mission_is_failed
				}
				check_variable = {
					decision_status = global.scripted_decision_default_status
				}
			}
		}
		set_variable = {
			days_mission_timeout_left = days_mission_timeout
		}
		set_variable = {
			decision_status = global.scripted_mission_is_active
		}
	}
}
cancel_scripted_decision = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					decision_status = global.scripted_decision_default_status
				}
			}
		}
		set_variable = {
			decision_status = global.scripted_decision_default_status
		}
		scripted_decision_start_re_enable = yes
		meta_effect = {
			text = {
				owner = {
					[t_decision]_cancel_effect = yes
				}
			}
			t_decision = "[?THIS.GetName]"
		}
	}
}

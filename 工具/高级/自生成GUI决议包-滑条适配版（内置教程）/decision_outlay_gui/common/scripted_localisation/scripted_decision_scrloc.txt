defined_text = {
    name = GetScriptedDecisionTimer
    text = {
        trigger = {
            var:scripted_decision_entry = {
                OR = {
                    check_variable = {
                        decision_status = global.scripted_decision_is_removing
                    }
                    check_variable = {
                        decision_status = global.scripted_mission_is_active
                    }
                    AND = {
                        scripted_decision_is_mission = yes
                        check_variable = {
                            decision_status = global.scripted_decision_default_status
                        }
                    }
                }
            }
            set_temp_variable = {
                t_timer = 0
            }
            var:scripted_decision_entry = {
                if = {
                    limit = {
                        check_variable = {
                            decision_status = global.scripted_decision_is_removing
                        }
                    }
                    add_to_temp_variable = {
                        t_timer = days_remove_left
                    }
                }
                if = {
                    limit = {
                        check_variable = {
                            decision_status = global.scripted_mission_is_active
                        }
                    }
                    add_to_temp_variable = {
                        t_timer = days_mission_timeout_left
                    }
                }
                if = {
                    limit = {
                        check_variable = {
                            decision_status = global.scripted_decision_default_status
                        }
                    }
                    add_to_temp_variable = {
                        t_timer = days_mission_timeout
                    }
                }
            }
        }
        localization_key = scripted_decision_timer_text
    }
    text = {
        localization_key = ""
    }
}

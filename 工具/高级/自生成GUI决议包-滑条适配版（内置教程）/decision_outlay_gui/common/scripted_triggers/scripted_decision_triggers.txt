sort_scripted_decision_category_check_prev_is_good_compare = {
	set_temp_variable = {
		t_decision_2 = scripted_decision_category_temp_list^k
	}
	set_temp_variable = {
		t_index_1 = k
	}
	add_to_temp_variable = {
		t_index_1 = 1
	}
	set_temp_variable = {
		t_decision_1 = scripted_decision_category_temp_list^t_index_1
	}

	set_temp_variable = {
		t_priority_1 = 0
	}
	set_temp_variable = {
		t_priority_2 = 0
	}
	var:t_decision_1 = {
		add_to_temp_variable = {
			t_priority_1 = priority
		}
	}
	var:t_decision_2 = {
		add_to_temp_variable = {
			t_priority_2 = priority
		}
	}
	if = {
		limit = {
			check_variable = {
				t_priority_1 = t_priority_2
			}
		}
		var:t_decision_1 = {
			if = {
				limit = {
					meta_trigger = {
						text = {
							[t_category]_visible = yes
						}
						t_category = "[?THIS.GetName]"
					}
				}
				set_temp_variable = {
					t_priority_1 = 0
				}
			}
			else = {
				set_temp_variable = {
					t_priority_1 = 1
				}
			}
		}
		var:t_decision_2 = {
			if = {
				limit = {
					meta_trigger = {
						text = {
							[t_category]_visible = yes
						}
						t_category = "[?THIS.GetName]"
					}
				}
				set_temp_variable = {
					t_priority_2 = 0
				}
			}
			else = {
				set_temp_variable = {
					t_priority_2 = 1
				}
			}
			##此处相反：数字小的优先级高
		}
		check_variable = {
			t_priority_1 < t_priority_2
		}
		##因此后面的数字不能比前面的小
	}
	else = {
		check_variable = {
			t_priority_1 > t_priority_2
		}
	}
}
sort_scripted_decision_category_array = {
	set_temp_variable = {
		t_sort_flag = 0
	}
	set_temp_variable = {
		t_max_index = scripted_decision_category_temp_list^num
	}
	subtract_from_temp_variable = {
		t_max_index = 1
	}
	##进行一次冒泡排序
	all_of = {
		array = scripted_decision_category_temp_list
		value = v1
		index = j	
		all_of = {
			array = scripted_decision_category_temp_list
			index = k
			value = v2
			if = {
				limit = {
					set_temp_variable = {
						t_limit = scripted_decision_category_temp_list^num
					}
					subtract_from_temp_variable = {
						t_limit = 1
					}
					subtract_from_temp_variable = {
						t_limit = j
					}
					check_variable = {
						k < t_limit
					}
				}
				if = {
					limit = {
						sort_scripted_decision_category_check_prev_is_good_compare = yes
					}
					set_temp_variable = {
						t_index_1 = k
					}
					add_to_temp_variable = {
						t_index_1 = 1
					}
					set_temp_variable = {
						t_temp = scripted_decision_category_temp_list^k
					}
					set_temp_variable = {
						scripted_decision_category_temp_list^k = scripted_decision_category_temp_list^t_index_1
					}
					set_temp_variable = {
						scripted_decision_category_temp_list^t_index_1 = t_temp
					}
					always = yes
				}
				always = yes
			}
			always = yes
		}
		always = yes
	}
}
sort_scripted_decision_check_prev_is_good_compare = {
	set_temp_variable = {
		t_decision_2 = scripted_decision_temp_list^k
	}
	set_temp_variable = {
		t_index_1 = k
	}
	add_to_temp_variable = {
		t_index_1 = 1
	}
	set_temp_variable = {
		t_decision_1 = scripted_decision_temp_list^t_index_1
	}

	set_temp_variable = {
		t_priority_1 = 0
	}
	set_temp_variable = {
		t_priority_2 = 0
	}
	var:t_decision_1 = {
		add_to_temp_variable = {
			t_priority_1 = priority
		}
	}
	var:t_decision_2 = {
		add_to_temp_variable = {
			t_priority_2 = priority
		}
	}
	if = {
		limit = {
			check_variable = {
				t_priority_1 = t_priority_2
			}
		}
		var:t_decision_1 = {
			if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_default_status
					}
					NOT = {
						scripted_decision_is_mission = yes
					}
					##正常的决议
				}
				set_temp_variable = {
					t_priority_1 = 0
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_mission_is_active
					}
					##激活了的任务
				}
				set_temp_variable = {
					t_priority_1 = 1
				}
			}
			else_if = {
				limit = {
					OR = {
						check_variable = {
							decision_status = global.scripted_decision_is_removing
						}
						check_variable = {
							decision_status = global.scripted_decision_has_removed
						}
					}
					##移除了或者移除中的任务
				}
				set_temp_variable = {
					t_priority_1 = 2
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_default_status
					}
					NOT = {
						scripted_decision_is_mission = yes
					}
					##未开始的决议
				}
				set_temp_variable = {
					t_priority_1 = 3
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_mission_is_failed
					}
					##失败了的决议
				}
				set_temp_variable = {
					t_priority_1 = 4
				}
			}
		}
		var:t_decision_2 = {
			if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_default_status
					}
					NOT = {
						scripted_decision_is_mission = yes
					}
					##正常的决议
				}
				set_temp_variable = {
					t_priority_2 = 0
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_mission_is_active
					}
					##激活了的任务
				}
				set_temp_variable = {
					t_priority_2 = 1
				}
			}
			else_if = {
				limit = {
					OR = {
						check_variable = {
							decision_status = global.scripted_decision_is_removing
						}
						check_variable = {
							decision_status = global.scripted_decision_has_removed
						}
					}
					##移除了或者移除中的任务
				}
				set_temp_variable = {
					t_priority_2 = 2
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_decision_default_status
					}
					NOT = {
						scripted_decision_is_mission = yes
					}
					##未开始的决议
				}
				set_temp_variable = {
					t_priority_2 = 3
				}
			}
			else_if = {
				limit = {
					check_variable = {
						decision_status = global.scripted_mission_is_failed
					}
					##失败了的决议
				}
				set_temp_variable = {
					t_priority_2 = 4
				}
			}
		}
		check_variable = {
			t_priority_1 < t_priority_2
		}
	}
	else = {
		check_variable = {
			t_priority_1 > t_priority_2
		}
	}
}
